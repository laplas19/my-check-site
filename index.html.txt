    function sendToWhatsapp() {
        // УКАЖИ СВОЙ НОМЕР ЗДЕСЬ (без плюса, например 49123456789)
        const phoneNumber = "49123456789"; 
        
        const banks = document.getElementById('banks').value || "Keine Angabe";
        const schufa = document.getElementById('schufa').value;

        const text = `Hallo! Ich interessiere mich für Ihr Angebot.
Hier sind meine Details:
- Bereits eröffnete Banken: ${banks}
- SCHUFA-Status: ${schufa}`;

        const encodedText = encodeURIComponent(text);
        const url = `https://wa.me/${phoneNumber}?text=${encodedText}`;
        
        window.open(url, '_blank');
    }
